                        if (data === "Suspendido") return '<span class="badge bg-warning text-dark">Suspendido</span>';

        function NuevoEstudiante() {

            var estudianteData = {
                Matricula: $("#txtMatricula").val().trim(),
                Nombres: $("#txtNombres").val().trim(),
                Apellidos: $("#txtApellidos").val().trim(),
                FechaNacimiento: $("#txtFechaNacimiento").val(),
                DocumentoIdentidad: $("#txtDocumento").val().trim(),
                CorreoInstitucional: $("#txtCorreo").val().trim(),
                Telefono: $("#txtTelefono").val().trim() || null // Si está vacío, envía null
            };

            // 2. Validación básica de campos obligatorios
            if (!estudianteData.Matricula || !estudianteData.Nombres || !estudianteData.Apellidos || !estudianteData.FechaNacimiento || !estudianteData.DocumentoIdentidad || !estudianteData.CorreoInstitucional) {
                alert("Por favor, complete todos los campos obligatorios del formulario.");
                return;
            }

            // 3. Deshabilitar el botón temporalmente para evitar doble clic erróneo
            $("#btnGuardar").prop("disabled", true).text("Guardando...");

            // 4. Petición AJAX POST hacia tu API de .NET
            $.ajax({
                url: "http://localhost:46533/api/values", // Asegúrate de usar el puerto correcto
                type: "POST",
                contentType: "application/json; charset=utf-8",
                dataType: "json",
                data: JSON.stringify(estudianteData), // Convierte el objeto JS a texto JSON string
                success: function (response) {
                    alert("¡Éxito! El estudiante ha sido registrado correctamente.");

                    // Cerrar la ventana modal de forma limpia usando Bootstrap 5
                    var modalElement = document.getElementById('clienteModal');
                    var modalInstance = bootstrap.Modal.getInstance(modalElement);
                    if (modalInstance) {
                        modalInstance.hide();
                    }

                    // Limpiar el formulario para un próximo registro
                    $("#txtMatricula, #txtNombres, #txtApellidos, #txtFechaNacimiento, #txtDocumento, #txtCorreo, #txtTelefono").val("");

                    // 🔄 REFRESCAR LA DATATABLE AUTOMÁTICAMENTE
                    // Si tu tabla de estudiantes se llama 'tablaEstudiantes', esto recarga los datos sin parpadear la pantalla
                    if ($.fn.DataTable.isDataTable('#tablaEstudiantes')) {
                        $('#tablaEstudiantes').DataTable().ajax.reload();
                    }
                },
                error: function (xhr, status, error) {
                    // Si el backend responde con un BadRequest() u otro error de validación
                    var mensajeError = xhr.responseJSON ? xhr.responseJSON.Message : "No se pudo conectar con el servidor.";
                    alert("Error al guardar: " + mensajeError);
                },
                complete: function () {
                    // Al finalizar (con éxito o error), volvemos a activar el botón de guardado
                    $("#btnGuardar").prop("disabled", false).text("Guardar Estudiante");
                }
            });
    
        }
