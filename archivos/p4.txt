CREATE PROCEDURE sp_ValidarUsuario
    @usuario VARCHAR(100),
    @PasswordHash VARCHAR(255) -- La contraseña ya debe venir cifrada desde la app
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (SELECT 1 FROM usuario WHERE usuario = @usuario AND     CONVERT(VARCHAR(300), DECRYPTBYPASSPHRASE('mipass', contrasena)) = @PasswordHash AND ideestado = 1)
    BEGIN
    select * from usuario where usuario=@usuario
    END
    ELSE
    BEGIN
        RAISERROR('Credenciales inválidas o usuario inactivo.', 16, 1);
    END
END

using Microsoft.AspNetCore.Mvc;
using Microsoft.Data.SqlClient;
using System.Data;
using UsuarioApi.modelos;



// For more information on enabling Web API for empty projects, visit https://go.microsoft.com/fwlink/?LinkID=397860

namespace UsuarioApi.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class UsuarioController : ControllerBase
    {
        private readonly string _connectionString;

        public UsuarioController(IConfiguration configuration)
        {

            _connectionString = configuration.GetConnectionString("DefaultConnection");
                }

            [HttpPost]


        public IActionResult ValidarUsuario([FromBody] UsuarioRequest usu)
        {
            bool usuarioExiste = false;
            string nombreUsuario = string.Empty;

            if (usu == null || string.IsNullOrEmpty(usu.Usuario) || string.IsNullOrEmpty(usu.Contraseña))
            {
                return BadRequest("Datos Invalidos");
            }

            try
            {
                // El 'using' asegura que la conexión se cierre sola al llegar al final del método
                using SqlConnection con = new SqlConnection(_connectionString);
                using SqlCommand cmd = new SqlCommand("sp_ValidarUsuario", con);

                cmd.CommandType = CommandType.StoredProcedure;

                cmd.Parameters.AddWithValue("@usuario", usu.Usuario);
                cmd.Parameters.AddWithValue("@PasswordHash", usu.Contraseña);

                con.Open();

                // El 'using' aquí asegura que el lector de datos se cierre inmediatamente después de usarse
                using SqlDataReader reader = cmd.ExecuteReader();

                if (reader.Read()) // Si el procedimiento devuelve al menos una fila, las credenciales son correctas
                {
                    usuarioExiste = true;
                    // Se lee la columna "usuario" que devuelve tu SP
                    nombreUsuario = reader["usuario"].ToString();
                }

                if (usuarioExiste)
                {
                    // Modificado para devolver 'nombreUsuario' (el que vino de la BD) en lugar de 'usu.Usuario'
                    return Ok(new { Mensaje = "Acceso concedido", Usuario = nombreUsuario });
                }
                else
                {
                    return Unauthorized("Usuario o contraseña incorrectos.");
                }
            }
            catch (Exception ex)
            {
                // En caso de que falle la conexión a la base de datos o el SP tenga un error
                return StatusCode(500, $"Error interno del servidor: {ex.Message}");
            }
        }



    }
    }
